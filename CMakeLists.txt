cmake_minimum_required(VERSION 3.20)
project(spsc_latency VERSION 1.0 LANGUAGES CXX)
option(SPSC_TSAN "Build with ThreadSanitizer" OFF)
option(SPSC_NATIVE "Tune for the build machine" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
include(CheckCXXCompilerFlag)
add_library(spsc INTERFACE)
target_include_directories(spsc INTERFACE include)
target_compile_options(spsc INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(SPSC_NATIVE)
  check_cxx_compiler_flag("-march=native" HAS_MARCH_NATIVE)
  if(HAS_MARCH_NATIVE)
    target_compile_options(spsc INTERFACE -march=native)
  endif()
endif()
if(SPSC_TSAN)
  target_compile_options(spsc INTERFACE -O1 -g -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(spsc INTERFACE -fsanitize=thread)
else()
  target_compile_options(spsc INTERFACE -O3)
endif()
foreach(target IN ITEMS spsc_tests spsc_bench)
  if(target STREQUAL "spsc_tests")
    add_executable(${target} tests/tests.cpp)
  else()
    add_executable(${target} bench/main.cpp)
  endif()
  target_include_directories(${target} PRIVATE bench)
  target_link_libraries(${target} PRIVATE spsc Threads::Threads)
  target_compile_definitions(${target} PRIVATE
    SPSC_BUILD_FLAGS="-std=c++20 ${CMAKE_CXX_FLAGS} $<$<CONFIG:Release>:${CMAKE_CXX_FLAGS_RELEASE}> $<JOIN:$<TARGET_PROPERTY:spsc,INTERFACE_COMPILE_OPTIONS>, >")
endforeach()
enable_testing()
add_test(NAME unit COMMAND spsc_tests --unit)
add_test(NAME stress COMMAND spsc_tests --items 1000000)
set_tests_properties(stress PROPERTIES TIMEOUT 180)
